# Automatically load overlays for detected cameras
camera_auto_detect=1

# Enable DRM VC4 V3D driver

# Don't have the firmware create an initial video= setting in cmdline.txt.

disable_fw_kms_setup=1

# Run in 64-bit mode
arm_64bit=1

# Disable compensation for displays with overscan
disable_overscan=1

# Run as fast as firmware / board allows
arm_boost=1

[cm4]
# Enable host mode on the 2711 built-in XHCI USB controller.
# This line should be removed if the legacy DWC2 controller is required
# (e.g. for USB device mode) or if USB support is not required.
otg_mode=1

[all]

# PiCAN-M: 16 MHz crystal, INT on GPIO 25 (per SK Pang/Copperhill user guide).
# A wrong oscillator value still lets the driver bind but skews bit timing.
# This overlay enables spi0 itself, so dtparam=spi=on is not needed.
dtoverlay=mcp2515-can0,oscillator=16000000,interrupt=25

# 400 kHz instead of the 100 kHz default. Every device on this bus (MCP9808, HTU31D,
# SHT41, ICM20948, ADXL345, MCP23017) and the LTC4307 buffer are rated for it, and it
# cuts a byte read from ~300us to ~75us. That matters for the rotary encoder: polling it
# was sampling slower than the knob generates edges, which showed up as missed
# quadrature states on fast spins.
dtparam=i2c_arm=on,i2c_arm_baudrate=400000

# --- Header UART on GPIO14/15 (/dev/serial0 -> ttyS0, mini UART) ---
# Left enabled, but NOT used by the GPS: GPIO15/pin10 is clamped low by something on the
# Pi side of the header, so no receiver can frame data there (measured: the pad drives
# both ways as an output, but as an input it stays low even with the internal pull-up,
# with nothing attached and with the hat's other candidates removed).
# PPS on GPIO23 was for the old ATGM336H GPS, which has been removed; the overlay is
# gone so GPIO23 is a free general-purpose pin again.
enable_uart=1

# --- GPS (u-blox M10) on uart5: GPIO12/pin32 = TX, GPIO13/pin33 = RX ---
# A real PL011 (fe201a00.serial), not the mini UART, and on pins that tested clean.
# The tty number the kernel hands out is not stable (Bluetooth's PL011 already takes
# ttyAMA1), so udev gives it the fixed name /dev/gps — see pi/etc/udev/rules.d/99-gps.rules.
dtoverlay=uart5
# --- Cooling fan enable on GPIO18/pin12 (driven high at boot = on; PWM controller may modulate) ---
gpio=18=op,dh
